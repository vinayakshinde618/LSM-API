-- tenants
CREATE TABLE tenants (
  id SERIAL PRIMARY KEY,
  tenant_name VARCHAR(255),
  email_id VARCHAR(255),
  phone VARCHAR(50),
  logo VARCHAR(255),
  address TEXT,
  is_active BOOLEAN,
  created_by INTEGER,
  updated_by INTEGER,
  "createdAt" TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  "updatedAt" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
